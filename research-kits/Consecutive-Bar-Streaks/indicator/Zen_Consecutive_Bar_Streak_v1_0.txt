//@version=6
indicator("Zen Consecutive Bar Streak v1.0", shorttitle="Zen CBS v1.0", overlay=true)

// --- Bar Direction ---
barDir = close > open ? 1 : close < open ? -1 : 0

// --- Streak Count ---
var int streak = 0

if barDir == 1
    streak := streak > 0 ? streak + 1 : 1
else if barDir == -1
    streak := streak < 0 ? streak - 1 : -1
else
    streak := 0

// --- Bar Colour (only colour when streak >= 2 or <= -2) ---
barcolor(streak >= 2 ? color.green : streak <= -2 ? color.red : na)

// --- Data Window Plots ---
plot(barDir, title="Bar Direction", display=display.data_window)
plot(streak, title="Streak",        display=display.data_window)
